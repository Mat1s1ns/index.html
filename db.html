CREATE TABLE Gramatas (
  GramataID   TEXT PRIMARY KEY,
  Autors      TEXT NOT NULL,
  Nosaukums   TEXT NOT NULL
);

CREATE TABLE Lietotaji (
  LietotajaID         TEXT PRIMARY KEY,
  LietotajaVards      TEXT NOT NULL,
  RezervacijasVesture TEXT DEFAULT ''
);

CREATE TABLE Rezervacija (
  RezervacijasID TEXT PRIMARY KEY,
  LietotajaID    TEXT REFERENCES Lietotaji(LietotajaID),
  GramataID      TEXT REFERENCES Gramatas(GramataID),
  Datums         TEXT
);
